# Módulo `fluxo_b_paredes` (S04-T13): biblioteca C++ pura `paredes` (a T14 liga o executável Qt nela) + ponte nanobind. Liga pelo IfcOpenShellConfig.cmake do conda, sem a geometria.
cmake_minimum_required(VERSION 3.20)
project(fluxo_b_paredes LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()
if(NOT DEFINED ENV{CONDA_PREFIX})
  message(FATAL_ERROR "CONDA_PREFIX não definido: ative o ambiente openbim")
endif()
file(TO_CMAKE_PATH "$ENV{CONDA_PREFIX}" CONDA)
list(APPEND CMAKE_PREFIX_PATH "${CONDA}" "${CONDA}/Library")
find_package(Python 3.12 REQUIRED COMPONENTS Interpreter Development.Module)
execute_process(COMMAND "${Python_EXECUTABLE}" -m nanobind --cmake_dir OUTPUT_VARIABLE NB_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
list(APPEND CMAKE_PREFIX_PATH "${NB_DIR}")
find_package(nanobind CONFIG REQUIRED)
find_package(IfcOpenShell CONFIG REQUIRED)
if(NOT OBIM_SAIDA)
  message(FATAL_ERROR "OBIM_SAIDA não definido: passe -DOBIM_SAIDA=<pasta de saída> (o build.sh faz isso)")
endif()
file(TO_CMAKE_PATH "${OBIM_SAIDA}" OBIM_SAIDA)
if(MSVC)
  add_compile_options(/utf-8)
endif()
add_library(paredes STATIC paredes.cpp)
set_target_properties(paredes PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(paredes PUBLIC IfcOpenShell::IfcParse IfcOpenShell::parse_schema_ifc4)
nanobind_add_module(fluxo_b_paredes ponte.cpp)
target_link_libraries(fluxo_b_paredes PRIVATE paredes)
set_target_properties(fluxo_b_paredes PROPERTIES LIBRARY_OUTPUT_DIRECTORY "$<1:${OBIM_SAIDA}>")  # saída plana, sem subpasta Release
if(WIN32)
  target_link_directories(paredes PUBLIC "${CONDA}/Library/lib")  # o Boost aparece nas dependências só pelo nome (boost_*.lib)
endif()
# Executável com janela (S04-T14c): camada C++ pura do núcleo (`obimcore_nucleo`, sem Python; o módulo `_obimcore` não entra: EXCLUDE_FROM_ALL) + `paredes`.
find_package(Qt6 COMPONENTS Core Gui Widgets GuiPrivate REQUIRED)  # antes do add_subdirectory: os alvos importados têm de valer também aqui
find_package(OpenCASCADE CONFIG REQUIRED)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../obimcore" obimcore_nucleo EXCLUDE_FROM_ALL)
add_executable(fluxo_b_janela main.cpp)
target_link_libraries(fluxo_b_janela PRIVATE paredes obimcore_nucleo)
set_target_properties(fluxo_b_janela PROPERTIES RUNTIME_OUTPUT_DIRECTORY "$<1:${OBIM_SAIDA}>")  # saída plana, ao lado do módulo
if(WIN32)
  target_link_libraries(fluxo_b_janela PRIVATE psapi)
endif()
# Oráculo da S04-T14a: T threads C++ criam o 1o `Paredes` do processo juntas (frio_teste.cpp; o implementador não o edita). Sai ao lado do módulo.
add_executable(fluxo_b_frio frio_teste.cpp)
target_link_libraries(fluxo_b_frio PRIVATE paredes)
set_target_properties(fluxo_b_frio PROPERTIES RUNTIME_OUTPUT_DIRECTORY "$<1:${OBIM_SAIDA}>")
